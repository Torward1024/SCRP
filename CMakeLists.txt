cmake_minimum_required(VERSION 3.16)
project(SCRP VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(SCRP_WITH_SDL2 "Build SDL2 presentation" ON)
option(SCRP_WITH_PHYSICS3D "Build optional Jolt physics and its C17 API" OFF)
option(SCRP_WITH_RENDER3D "Build optional OpenGL 3.3 presentation and its C17 API" OFF)
add_library(scrp_core src/Json.cpp src/Localization.cpp src/Vfs.cpp src/IndexedImage.cpp src/Config.cpp src/Xml.cpp src/Anim.cpp src/Grid.cpp src/FlowField.cpp src/PathFinder.cpp src/SaveStore.cpp src/Rnc.cpp src/PackWriter.cpp)
add_library(SCRP::Core ALIAS scrp_core)
target_sources(scrp_core PRIVATE src/Math3D.cpp src/Mesh.cpp)
target_include_directories(scrp_core PUBLIC include)
target_compile_features(scrp_core PUBLIC cxx_std_17)
if(SCRP_WITH_PHYSICS3D)
  include(cmake/Physics3D.cmake)
endif()
if(SCRP_WITH_SDL2)
  find_package(SDL2 REQUIRED)
  option(SCRP_USE_SDL_IMAGE "Enable PNG sprite loading" OFF)
  option(SCRP_USE_SDL_MIXER "Enable SDL_mixer audio" OFF)
  add_library(scrp_sdl2 src/PixelWindow.cpp src/Assets.cpp src/Gfx.cpp src/MicroFont.cpp src/BitmapFont.cpp src/Lighting.cpp src/Decals.cpp src/Particles.cpp src/Window.cpp)
  add_library(scrp_audio src/Audio.cpp)
  add_library(SCRP::Audio ALIAS scrp_audio)
  target_link_libraries(scrp_audio PUBLIC SCRP::Core SDL2::SDL2)
  if(SCRP_USE_SDL_IMAGE)
    find_package(SDL2_image REQUIRED)
    target_compile_definitions(scrp_sdl2 PRIVATE SCRP_USE_SDL_IMAGE)
    target_link_libraries(scrp_sdl2 PRIVATE SDL2_image::SDL2_image)
  endif()
  if(SCRP_USE_SDL_MIXER)
    find_package(SDL2_mixer REQUIRED)
    target_compile_definitions(scrp_audio PRIVATE SCRP_USE_SDL_MIXER)
    target_link_libraries(scrp_audio PRIVATE SDL2_mixer::SDL2_mixer)
  endif()
  add_library(SCRP::SDL2 ALIAS scrp_sdl2)
  target_link_libraries(scrp_sdl2 PUBLIC SCRP::Core SDL2::SDL2)
endif()
if(SCRP_WITH_RENDER3D)
  if(NOT SCRP_WITH_SDL2)
    message(FATAL_ERROR "SCRP_WITH_RENDER3D requires SCRP_WITH_SDL2")
  endif()
  find_package(OpenGL REQUIRED)
  add_library(scrp_render3d src/Render3D.cpp)
  add_library(SCRP::Render3D ALIAS scrp_render3d)
  target_link_libraries(scrp_render3d PUBLIC SCRP::Core PRIVATE SCRP::SDL2 OpenGL::GL)
  target_compile_definitions(scrp_render3d PRIVATE SDL_MAIN_HANDLED)
  option(SCRP_TEST_RENDER3D "Run 3D tests with a real OpenGL context (not SDL dummy)" OFF)
  if(BUILD_TESTING AND SCRP_TEST_RENDER3D)
    enable_language(C)
    add_executable(scrp_render3d_tests tests/render3d.c)
    target_link_libraries(scrp_render3d_tests PRIVATE SCRP::Render3D)
    set_target_properties(scrp_render3d_tests PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE CXX)
    add_test(NAME scrp_render3d_c_contract COMMAND scrp_render3d_tests)
  endif()
endif()
if(BUILD_TESTING)
  enable_language(C)
  add_executable(scrp_mesh_tests tests/mesh.c)
  target_link_libraries(scrp_mesh_tests PRIVATE SCRP::Core)
  set_target_properties(scrp_mesh_tests PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE CXX)
  add_test(NAME scrp_mesh_c_contract COMMAND scrp_mesh_tests)
  add_executable(scrp_tests tests/main.cpp)
  target_link_libraries(scrp_tests PRIVATE SCRP::Core)
  add_test(NAME scrp_core COMMAND scrp_tests)
  add_executable(scrp_runtime_tests tests/runtime.cpp)
  target_link_libraries(scrp_runtime_tests PRIVATE SCRP::Core)
  add_test(NAME scrp_runtime COMMAND scrp_runtime_tests)
  if(SCRP_WITH_SDL2)
    add_executable(scrp_sdl_tests tests/sdl.cpp)
    target_link_libraries(scrp_sdl_tests PRIVATE SCRP::SDL2 SCRP::Audio)
    target_compile_definitions(scrp_sdl_tests PRIVATE SDL_MAIN_HANDLED)
    if(SCRP_USE_SDL_MIXER)
      target_compile_definitions(scrp_sdl_tests PRIVATE SCRP_TEST_WITH_MIXER)
    endif()
    add_test(NAME scrp_sdl COMMAND scrp_sdl_tests)
    set_tests_properties(scrp_sdl PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endif()
endif()
