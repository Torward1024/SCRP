cmake_minimum_required(VERSION 3.16)
project(SCRP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(SCRP_WITH_SDL2 "Build SDL2 presentation" ON)
add_library(scrp_core src/Json.cpp src/Vfs.cpp src/IndexedImage.cpp)
add_library(SCRP::Core ALIAS scrp_core)
target_include_directories(scrp_core PUBLIC include)
target_compile_features(scrp_core PUBLIC cxx_std_17)
if(SCRP_WITH_SDL2)
  find_package(SDL2 REQUIRED)
  add_library(scrp_sdl2 src/PixelWindow.cpp)
  add_library(SCRP::SDL2 ALIAS scrp_sdl2)
  target_link_libraries(scrp_sdl2 PUBLIC SCRP::Core SDL2::SDL2)
endif()
if(BUILD_TESTING)
  add_executable(scrp_tests tests/main.cpp)
  target_link_libraries(scrp_tests PRIVATE SCRP::Core)
  add_test(NAME scrp_core COMMAND scrp_tests)
endif()
