cmake_minimum_required(VERSION 3.16)
project(SCRP VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(SCRP_WITH_SDL2 "Build SDL2 presentation" ON)
add_library(scrp_core src/Json.cpp src/Vfs.cpp src/IndexedImage.cpp src/Config.cpp src/Xml.cpp src/Anim.cpp src/Grid.cpp src/FlowField.cpp src/PathFinder.cpp src/SaveStore.cpp)
add_library(SCRP::Core ALIAS scrp_core)
target_include_directories(scrp_core PUBLIC include)
target_compile_features(scrp_core PUBLIC cxx_std_17)
if(SCRP_WITH_SDL2)
  find_package(SDL2 REQUIRED)
  option(SCRP_USE_SDL_IMAGE "Enable PNG sprite loading" OFF)
  option(SCRP_USE_SDL_MIXER "Enable SDL_mixer audio" OFF)
  add_library(scrp_sdl2 src/PixelWindow.cpp src/Assets.cpp src/Gfx.cpp src/MicroFont.cpp src/Lighting.cpp src/Decals.cpp src/Particles.cpp src/Window.cpp)
  add_library(scrp_audio src/Audio.cpp)
  add_library(SCRP::Audio ALIAS scrp_audio)
  target_link_libraries(scrp_audio PUBLIC SCRP::Core SDL2::SDL2)
  if(SCRP_USE_SDL_IMAGE)
    find_package(SDL2_image REQUIRED)
    target_compile_definitions(scrp_sdl2 PRIVATE SCRP_USE_SDL_IMAGE)
    target_link_libraries(scrp_sdl2 PRIVATE SDL2_image::SDL2_image)
  endif()
  if(SCRP_USE_SDL_MIXER)
    find_package(SDL2_mixer REQUIRED)
    target_compile_definitions(scrp_audio PRIVATE SCRP_USE_SDL_MIXER)
    target_link_libraries(scrp_audio PRIVATE SDL2_mixer::SDL2_mixer)
  endif()
  add_library(SCRP::SDL2 ALIAS scrp_sdl2)
  target_link_libraries(scrp_sdl2 PUBLIC SCRP::Core SDL2::SDL2)
endif()
if(BUILD_TESTING)
  add_executable(scrp_tests tests/main.cpp)
  target_link_libraries(scrp_tests PRIVATE SCRP::Core)
  add_test(NAME scrp_core COMMAND scrp_tests)
  add_executable(scrp_runtime_tests tests/runtime.cpp)
  target_link_libraries(scrp_runtime_tests PRIVATE SCRP::Core)
  add_test(NAME scrp_runtime COMMAND scrp_runtime_tests)
  if(SCRP_WITH_SDL2)
    add_executable(scrp_sdl_tests tests/sdl.cpp)
    target_link_libraries(scrp_sdl_tests PRIVATE SCRP::SDL2 SCRP::Audio)
    target_compile_definitions(scrp_sdl_tests PRIVATE SDL_MAIN_HANDLED)
    if(SCRP_USE_SDL_MIXER)
      target_compile_definitions(scrp_sdl_tests PRIVATE SCRP_TEST_WITH_MIXER)
    endif()
    add_test(NAME scrp_sdl COMMAND scrp_sdl_tests)
    set_tests_properties(scrp_sdl PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endif()
endif()
